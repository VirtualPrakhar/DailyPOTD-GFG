import java.util.*;

class Solution {
    public int maxPeople(int[] arr) {
        int n = arr.length;
        int[] leftGE = new int[n];
        int[] rightGE = new int[n];

        Stack<Integer> stack = new Stack<>();

        // Nearest greater or equal to LEFT
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && arr[stack.peek()] < arr[i]) {
                stack.pop();
            }
            leftGE[i] = stack.isEmpty() ? -1 : stack.peek();
            stack.push(i);
        }

        stack.clear();

        // Nearest greater or equal to RIGHT
        for (int i = n - 1; i >= 0; i--) {
            while (!stack.isEmpty() && arr[stack.peek()] < arr[i]) {
                stack.pop();
            }
            rightGE[i] = stack.isEmpty() ? n : stack.peek();
            stack.push(i);
        }

        int maxVisible = 0;

        for (int i = 0; i < n; i++) {
            int visible = (i - leftGE[i] - 1) + (rightGE[i] - i - 1) + 1;
            maxVisible = Math.max(maxVisible, visible);
        }

        return maxVisible;
    }
}

        # Nearest greater or equal to LEFT
        for i in range(n):
            while stack and arr[stack[-1]] < arr[i]:
                stack.pop()
            leftGE[i] = stack[-1] if stack else -1
            stack.append(i)

        stack.clear()

        # Nearest greater or equal to RIGHT

            while stack and arr[stack[-1]] < arr[i]:
                stack.pop()

            stack.append(i)

        ans = 0
        for i in range(n):
            visible = (i - leftGE[i] - 1) + (rightGE[i] - i - 1) + 1
            ans = max(ans, visible)

        return ans

        const rightGE = new Array(n);

        // Nearest greater or equal to LEFT
        for (let i = 0; i < n; i++) {
