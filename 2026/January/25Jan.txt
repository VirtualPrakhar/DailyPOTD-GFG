=========================Java Solution==================================
class Solution {
    int findWays(int n) {
        // Odd length cannot form valid parentheses
        if (n % 2 != 0) return 0;

        int pairs = n / 2;
        int[] dp = new int[pairs + 1];

        dp[0] = 1; // empty string
        dp[1] = 1; // "()"

        for (int i = 2; i <= pairs; i++) {
            for (int j = 0; j < i; j++) {
                dp[i] += dp[j] * dp[i - 1 - j];
            }
        }

        return dp[pairs];
    }
}


=========================Python Solution==============================
class Solution:
    def findWays(self, n: int) -> int:
        if n % 2 != 0:
            return 0

        pairs = n // 2
        dp = [0] * (pairs + 1)
        dp[0] = 1
        dp[1] = 1

        for i in range(2, pairs + 1):
            for j in range(i):
                dp[i] += dp[j] * dp[i - 1 - j]

        return dp[pairs]
