=========================Java Solution==================================
class Solution {
    public static ArrayList<ArrayList<Integer>> permuteDist(int[] arr) {
        ArrayList<ArrayList<Integer>> result = new ArrayList<>();
        boolean[] used = new boolean[arr.length];
        backtrack(arr, used, new ArrayList<>(), result);
        return result;
    }

    private static void backtrack(int[] arr, boolean[] used,
                                  ArrayList<Integer> current,
                                  ArrayList<ArrayList<Integer>> result) {
        if (current.size() == arr.length) {
            result.add(new ArrayList<>(current));
            return;
        }

        for (int i = 0; i < arr.length; i++) {
            if (used[i]) continue;

            used[i] = true;
            current.add(arr[i]);

            backtrack(arr, used, current, result);

            current.remove(current.size() - 1);
            used[i] = false;
        }
    }
}


=========================Python Solution==============================
class Solution:
    def permuteDist(self, arr):
        res = []
        used = [False] * len(arr)

        def backtrack(path):
            if len(path) == len(arr):
                res.append(path[:])
                return

            for i in range(len(arr)):
                if used[i]:
                    continue
                used[i] = True
                path.append(arr[i])
                backtrack(path)
                path.pop()
                used[i] = False

        backtrack([])
        return res


===========================C++ Solution==============================
class Solution {
public:
    vector<vector<int>> permuteDist(vector<int>& arr) {
        vector<vector<int>> res;
        vector<int> path;
        vector<bool> used(arr.size(), false);
        backtrack(arr, used, path, res);
        return res;
    }

    void backtrack(vector<int>& arr, vector<bool>& used,
                   vector<int>& path, vector<vector<int>>& res) {
        if (path.size() == arr.size()) {
            res.push_back(path);
            return;
        }

        for (int i = 0; i < arr.size(); i++) {
            if (used[i]) continue;

            used[i] = true;
            path.push_back(arr[i]);
            backtrack(arr, used, path, res);
            path.pop_back();
            used[i] = false;
        }
    }
};

class Solution {
    permuteDist(arr) {
        const res = [];
        const used = new Array(arr.length).fill(false);

        const backtrack = (path) => {
            if (path.length === arr.length) {
                res.push([...path]);
                return;
            }

            for (let i = 0; i < arr.length; i++) {
                if (used[i]) continue;

                used[i] = true;
                path.push(arr[i]);
                backtrack(path);
                path.pop();
                used[i] = false;
            }
        };

        backtrack([]);
        return res;
    }
}
